🌙 Kimi Vision AI — Handwritten OCR

Moonshot AI's Kimi reads handwriting far better than Tesseract — use it when sheet photos are hard to OCR
Why the script is back: Moonshot blocks browser CORS. A Google Apps Script web app calls Moonshot from Google’s servers, so the photo still works from this page.
① Recommended — Apps Script proxy: paste the …/exec URL (same deployment as cloud sync is fine). The Moonshot key lives in the script, not in the browser.
② Optional — browser key: paste a Moonshot key for a direct try first; if CORS blocks it, the proxy is used automatically.
③ No key / no script? Manual mode copies the photo and opens Kimi.com.
Full script: google-apps-script.gs in this repo. Model default: kimi-k3.

📋 Provider-agnostic Apps Script proxy (Gemini by default — or use the full google-apps-script.gs)
  1. Open script.google.com → New project → paste below → set GEMINI_API_KEY (free from AI Studio) — or MOONSHOT_API_KEY for Kimi.
  2. Deploy → New deployment → Web app → Execute as: Me · Who has access: Anyone.
  3. Paste the …/exec URL into the proxy field (or reuse the same URL as cloud sync).
var GEMINI_API_KEY = 'YOUR_GEMINI_KEY';   // free: aistudio.google.com/apikey
var MOONSHOT_API_KEY = 'YOUR_API_KEY';    // optional, paid

function doPost(e) {
  var body = JSON.parse(e.postData.contents);
  if (body.action !== 'kimiVision') {
    return ContentService.createTextOutput(JSON.stringify({success:false,error:'unknown action'})).setMimeType(ContentService.MimeType.JSON);
  }
  var base = String(body.base || 'https://generativelanguage.googleapis.com/v1beta').replace(/\/+$/, '').replace(/\/openai$/, '');
  var isGemini = base.indexOf('generativelanguage.googleapis.com') !== -1;
  var key = isGemini ? GEMINI_API_KEY : MOONSHOT_API_KEY;
  var model = body.model || (isGemini ? 'gemini-2.5-flash' : 'kimi-k3');
  var prompt = body.prompt || 'Extract all text from this image. Return ONLY the text.';
  if (isGemini) {
    var durl = String(body.image || '');
    var comma = durl.indexOf(',');
    var img = comma >= 0 ? durl.slice(comma + 1) : durl;
    var mm = /^data:(image\/[a-z+.-]+);/i.exec(durl);
    var res = UrlFetchApp.fetch(base + '/models/' + encodeURIComponent(model) + ':generateContent?key=' + encodeURIComponent(key), {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify({ contents: [{ parts: [
        { inline_data: { mime_type: mm ? mm[1] : 'image/jpeg', data: img } },
        { text: prompt }
      ] }] }),
      muteHttpExceptions: true
    });
    var gdata = JSON.parse(res.getContentText());
    if (gdata.candidates && gdata.candidates[0] && gdata.candidates[0].content) {
      var parts = gdata.candidates[0].content.parts || [];
      var gtext = parts.map(function (p) { return p.text || ''; }).join('');
      if (gtext) return ContentService.createTextOutput(JSON.stringify({success:true,text:gtext})).setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({success:false,error:gdata.error||gdata})).setMimeType(ContentService.MimeType.JSON);
  }
  var payload = {
    model: model,
    temperature: 0.1,
    messages: [{ role: 'user', content: [
      { type: 'image_url', image_url: { url: body.image } },
      { type: 'text', text: prompt }
    ]}]
  };
  var res = UrlFetchApp.fetch(base + '/chat/completions', {
    method: 'post',
    contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + key },
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  });
  var data = JSON.parse(res.getContentText());
  if (data.choices && data.choices[0]) {
    return ContentService.createTextOutput(JSON.stringify({success:true,text:data.choices[0].message.content})).setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({success:false,error:data.error||data})).setMimeType(ContentService.MimeType.JSON);
}
